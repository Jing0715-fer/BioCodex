    49→
    50→export function TaxonDetail({ id }: { id: string }) {
    51→  const { openTaxon, explore, goBack, compareIds, toggleCompare, openCompare } = useBioStore();
    52→  const { data, isLoading } = useTaxon(id);
    53→  const { data: tree } = useTree();
    54→  const [lightbox, setLightbox] = useState(false);
    55→
    56→  // 面包屑物种计数:从树 DTO 递归建 id → 物种数映射(与探索视图一致)
    57→  const speciesCountById = useMemo(() => {
    58→    const m = new Map<string, number>();
    59→    const walk = (nodes: TreeNodeDTO[]) => {
    60→      for (const n of nodes) {
    61→        if (!m.has(n.id)) m.set(n.id, n.sc);
    62→        if (n.ch?.length) walk(n.ch);
    63→      }
    64→    };
    65→    if (tree) walk(tree);
    66→    return m;
    67→  }, [tree]);
    68→
    69→  const links = useMemo(
    70→    () =>
    71→      data
    72→        ? buildDbLinks({
    73→            latinName: data.taxon.latinName,
    74→            chineseName: data.taxon.chineseName,
    75→            ncbiTaxId: data.taxon.ncbiTaxId,
    76→            rank: data.taxon.rank,
    77→            kingdomPath: data.taxon.kingdomPath,
    78→          })
    79→        : [],
    80→    [data]
    81→  );
    82→
    83→  const grouped = useMemo(() => {
    84→    const g = new Map<string, typeof links>();
    85→    for (const l of links) {
    86→      if (!g.has(l.group)) g.set(l.group, []);
    87→      g.get(l.group)!.push(l);
    88→    }
    89→    return g;
    90→  }, [links]);
    91→
    92→  // 上一/下一物种(hooks 必须在 early return 之前计算)
    93→  const siblings = data?.siblings || [];
    94→  const idx = siblings.findIndex((s) => s.id === id);
    95→  const prev = idx > 0 ? siblings[idx - 1] : null;
    96→  const next = idx >= 0 && idx < siblings.length - 1 ? siblings[idx + 1] : null;
    97→
    98→  // 键盘导航:←/→ 切换同属上/下一物种(输入框聚焦时忽略);Esc 关灯箱
    99→  useEffect(() => {
   100→    const onKey = (e: KeyboardEvent) => {
   101→      const tag = (e.target as HTMLElement)?.tagName;
   102→      if (tag === "INPUT" || tag === "TEXTAREA" || (e.target as HTMLElement)?.isContentEditable) return;
   103→      if (e.key === "Escape") setLightbox(false);
   104→      if (e.key === "ArrowLeft" && prev) openTaxon(prev.id);
   105→      if (e.key === "ArrowRight" && next) openTaxon(next.id);
   106→    };
   107→    window.addEventListener("keydown", onKey);
   108→    return () => window.removeEventListener("keydown", onKey);
   109→  }, [prev, next, openTaxon]);
   110→
   111→  if (isLoading || !data?.success) {
   112→    return (
   113→      <div className="mx-auto max-w-[1400px] px-4 py-8 sm:px-6">
   114→        <div className="flex flex-col gap-4">
   115→          <Skeleton className="h-5 w-64" />
   116→          <Skeleton className="h-72 w-full rounded-xl" />
   117→          <div className="grid gap-4 lg:grid-cols-3">
   118→            <div className="space-y-4 lg:col-span-2">
   119→              <Skeleton className="h-40 w-full rounded-xl" />
   120→            </div>
   121→            <Skeleton className="h-40 rounded-xl" />
   122→          </div>
   123→        </div>
   124→      </div>
   125→    );
   126→  }
   127→
   128→  const { taxon, lineage, children, counts } = data;
   129→  const theme = KINGDOM_THEME[taxon.kingdom] || KINGDOM_THEME.Animalia;
   130→  const isSpecies = taxon.rank === "species";
   131→  const iucn = taxon.conservation ? IUCN_INFO[taxon.conservation] : null;
   132→  const tags = taxon.tags || [];
   133→  // 同属近亲(排除自身)
   134→  const relatedSpecies = siblings.filter(
   135→    (s) => s.rank === "species" && s.id !== taxon.id
   136→  );
   137→  const genusName = lineage[lineage.length - 1]?.chineseName;
   138→
   139→  return (
   140→    <div className="mx-auto w-full max-w-[1400px] px-4 pb-8 pt-4 sm:px-6">
   141→      {/* 面包屑 */}
   142→      <nav className="flex flex-wrap items-center gap-1 text-xs text-muted-foreground" aria-label="分类路径">
   143→        <button className="hover:text-primary" onClick={() => explore(null)}>
   144→          生命之树
   145→        </button>
   146→        {lineage.map((l) => {
   147→          const sc = speciesCountById.get(l.id);
   148→          return (
   149→            <span key={l.id} className="flex items-center gap-1">
   150→              <ChevronRight className="h-3 w-3 text-muted-foreground/50" />
   151→              <button
   152→                className="flex max-w-32 items-center gap-1 truncate hover:text-primary"
   153→                onClick={() => explore(l.id)}
   154→                title={`${l.chineseName}${sc ? ` · ${sc} 物种` : ""}`}
   155→              >
   156→                <span className="truncate">{l.chineseName}</span>
   157→                {sc != null && sc > 0 && (
   158→                  <span className="shrink-0 rounded-full bg-foreground/8 px-1.5 text-[9px] font-semibold tabular-nums text-muted-foreground/80 dark:bg-foreground/15">
   159→                    {sc}
   160→                  </span>
   161→                )}
   162→              </button>
   163→            </span>
   164→          );
   165→        })}
   166→        <ChevronRight className="h-3 w-3 text-muted-foreground/50" />
   167→        <span className="font-medium text-foreground">{taxon.chineseName}</span>
   168→      </nav>
   169→
   170→      {/* ====== 主图区 ====== */}
   171→      <div className="reveal-up relative mt-3 overflow-hidden rounded-2xl border border-foreground/10 shadow-md">
   172→        {taxon.image ? (
   173→          <button
   174→            onClick={() => setLightbox(true)}
   175→            aria-label="放大查看物种插图"
   176→            className="group/img absolute inset-0 h-full w-full cursor-zoom-in"
   177→          >
   178→            <img
   179→              src={taxon.image}
   180→              alt={`${taxon.chineseName}(${taxon.latinName})博物学插图`}
   181→              className="img-fade-in h-[320px] w-full object-cover transition-transform duration-500 group-hover/img:scale-[1.02] sm:h-[420px]"
   182→            />
   183→          </button>
   184→        ) : (
   185→          <TaxaPlaceholder latinName={taxon.latinName} kingdom={taxon.kingdom} big className="h-[320px] w-full sm:h-[420px]" />
   186→        )}
   187→        <div className="pointer-events-none absolute inset-0 bg-gradient-to-t from-black/70 via-black/15 to-transparent" />
   188→        {/* 放大提示角标(有图时) */}
   189→        {taxon.image && (
   190→          <span className="pointer-events-none absolute bottom-3 right-3 flex items-center gap-1 rounded-full bg-black/50 px-2.5 py-1 text-[11px] font-medium text-white/90 opacity-0 backdrop-blur-sm transition-opacity duration-300 group-hover/img:opacity-100 sm:bottom-5 sm:right-5">
   191→            <ZoomIn className="h-3 w-3" />
   192→            点击放大
   193→          </span>
   194→        )}
   195→        {/* 右上角操作:加入对比(仅物种) */}
   196→        {isSpecies && (
   197→          <div className="absolute right-3 top-3 flex items-center gap-2">
   198→            {(() => {
   199→              const inCompare = compareIds.includes(taxon.id);
   200→              const full = !inCompare && compareIds.length >= MAX_COMPARE;
   201→              return (
   202→                <>
   203→                  <button
   204→                    onClick={() => {
   205→                      const res = toggleCompare(taxon.id);
   206→                      if (res === "added")
   207→                        toast.success(`已加入对比:${taxon.chineseName}`, {
   208→                          description: `托盘 ${compareIds.length + 1}/${MAX_COMPARE},选满 2 个即可开始`,
   209→                        });
   210→                      else if (res === "removed") toast.info(`已移出对比:${taxon.chineseName}`);
   211→                      else toast.warning("对比托盘已满(最多 3 个)", { description: "可先开始对比或移除一个物种" });
   212→                    }}
   213→                    disabled={full}
   214→                    aria-label={inCompare ? `移出对比:${taxon.chineseName}` : `加入对比:${taxon.chineseName}`}
   215→                    className={cn(
   216→                      "flex items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-bold shadow transition-all",
   217→                      inCompare
   218→                        ? "bg-primary text-primary-foreground"
   219→                        : "bg-black/50 text-white backdrop-blur-sm hover:bg-black/70",
   220→                      full && "cursor-not-allowed opacity-50"
   221→                    )}
   222→                  >
   223→                    {inCompare ? <Check className="h-3.5 w-3.5" /> : <GitCompareArrows className="h-3.5 w-3.5" />}
   224→                    {inCompare ? "已加入对比" : "加入对比"}
   225→                  </button>
   226→                  {compareIds.length >= 2 && (
   227→                    <button
   228→                      onClick={openCompare}
   229→                      className="hidden items-center gap-1.5 rounded-full bg-white/20 px-3 py-1.5 text-xs font-bold text-white shadow backdrop-blur-sm transition-colors hover:bg-white/35 sm:flex"
   230→                    >
   231→                      对比 {compareIds.length} 个物种
   232→                    </button>
   233→                  )}
   234→                </>
   235→              );
   236→            })()}
   237→          </div>
   238→        )}
   239→        {/* 标题叠层 */}
   240→        <div className="absolute bottom-0 left-0 right-0 p-5 sm:p-8">
   241→          <div className="flex flex-wrap items-center gap-2">
   242→            <span
   243→              className="flex items-center gap-1.5 rounded-full px-3 py-1 text-xs font-bold text-white shadow"
   244→              style={{ background: theme.color }}
   245→            >
   246→              <KingdomIcon kingdom={taxon.kingdom} className="h-3.5 w-3.5" />
   247→              {theme.name}
   248→            </span>
   249→            <span className="rounded-sm bg-white/20 px-2 py-0.5 text-xs font-medium text-white backdrop-blur-sm">
   250→              {rankLabel(taxon.rank)}
   251→            </span>
   252→            {iucn && (
   253→              <span className={cn("rounded-sm px-2 py-0.5 text-xs font-bold text-white shadow", iucn.bg)}>
   254→                IUCN {iucn.label}
   255→              </span>
   256→            )}
   257→            {tags.includes("flagship") && (
   258→              <span className="flex items-center gap-1 rounded-sm bg-amber-500/85 px-2 py-0.5 text-xs font-bold text-white shadow">
   259→                <Star className="h-3 w-3" />
   260→                旗舰物种
   261→              </span>
   262→            )}
   263→            {tags.includes("模式生物") && (
   264→              <span className="flex items-center gap-1 rounded-sm bg-teal-600/85 px-2 py-0.5 text-xs font-bold text-white shadow">
   265→                <Microscope className="h-3 w-3" />
   266→                模式生物
   267→              </span>
   268→            )}
   269→            {tags.includes("入侵物种") && (
   270→              <span className="rounded-sm bg-zinc-700/85 px-2 py-0.5 text-xs font-bold text-white shadow">
   271→                入侵物种
   272→              </span>
   273→            )}
   274→          </div>
   275→          <h1 className="mt-3 font-display text-3xl font-bold text-white drop-shadow-md sm:text-5xl">
   276→            {taxon.chineseName}
   277→          </h1>
   278→          <div className="mt-1.5 flex flex-wrap items-baseline gap-3">
   279→            <span className="latin text-lg text-white/95 drop-shadow sm:text-2xl">{taxon.latinName}</span>
   280→            {taxon.authority && (
   281→              <span className="text-sm text-white/75 drop-shadow">{taxon.authority}</span>
   282→            )}
   283→            {taxon.ncbiTaxId && (
   284→              <span className="latin rounded-sm bg-white/15 px-2 py-0.5 text-xs text-white/90 backdrop-blur-sm">
   285→                NCBI txid{taxon.ncbiTaxId}
   286→              </span>
   287→            )}
   288→          </div>
   289→        </div>
   290→      </div>
   291→
   292→      {/* ====== 内容 + 侧栏 ====== */}
   293→      <div className="mt-5 grid gap-5 lg:grid-cols-[1fr_360px]">
   294→        {/* 左列:正文 */}
   295→        <div className="min-w-0 space-y-5">
   296→          {taxon.description && (
   297→            <SectionCard title={isSpecies ? "物种描述" : "类群概述"} icon={BookOpen}>
   298→              {taxon.description}
   299→            </SectionCard>
   300→          )}
   301→
   302→          {(taxon.morphology || taxon.habitat || taxon.distribution) && (
   303→            <SectionCard title="形态、生境与分布" icon={Leaf}>
   304→              <dl className="space-y-4">
   305→                {taxon.morphology && (
   306→                  <div>
   307→                    <dt className="mb-1 text-xs font-bold tracking-widest text-muted-foreground">形态识别</dt>
   308→                    <dd>{taxon.morphology}</dd>
   309→                  </div>
   310→                )}
   311→                {taxon.habitat && (
   312→                  <div>
   313→                    <dt className="mb-1 text-xs font-bold tracking-widest text-muted-foreground">生境</dt>
   314→                    <dd>{taxon.habitat}</dd>
   315→                  </div>
   316→                )}
   317→                {taxon.distribution && (
   318→                  <div>
   319→                    <dt className="mb-1 text-xs font-bold tracking-widest text-muted-foreground">分布</dt>
   320→                    <dd className="flex items-start gap-2">
   321→                      <MapPin className="mt-1.5 h-4 w-4 shrink-0 text-primary/70" />
   322→                      {taxon.distribution}
   323→                    </dd>
   324→                  </div>
   325→                )}
   326→              </dl>
   327→            </SectionCard>
   328→          )}
   329→
   330→          {iucn && taxon.conservation && (
   331→            <SectionCard title="保护状况" icon={Shield}>
   332→              <div className="flex flex-wrap items-center gap-3">
   333→                <span className={cn("rounded-md px-3 py-1.5 font-display text-lg font-bold text-white", iucn.bg)}>
   334→                  {taxon.conservation}
   335→                </span>
   336→                <div>
   337→                  <p className="font-semibold text-foreground">{iucn.full}</p>
   338→                  <p className="mt-0.5 text-xs leading-5 text-muted-foreground">
   339→                    依据 IUCN 红色名录等级:极危(CR)与濒危(EN)物种面临野外灭绝的高风险;
   340→                    图鉴中 {data.counts.speciesCount >= 0 && "收录的该等级物种均附红色名录直链,可查最新评估。"}
   341→                  </p>
   342→                </div>
   343→              </div>
   344→            </SectionCard>
   345→          )}
   346→
   347→          {/* 同属近亲 */}
   348→          {isSpecies && relatedSpecies.length > 0 && (
   349→            <SectionCard
   350→              title={`${genusName || "同属"}的其它成员(${relatedSpecies.length})`}
   351→              icon={Sparkles}
   352→            >
   353→              <div className="nh-scroll -mx-1 flex gap-3 overflow-x-auto px-1 pb-2">
   354→                {relatedSpecies.map((r) => (
   355→                  <button
   356→                    key={r.id}
   357→                    onClick={() => openTaxon(r.id)}
   358→                    className="group relative w-40 shrink-0 overflow-hidden rounded-lg border border-foreground/10 bg-muted/30 text-left transition-all hover:border-primary/40 hover:shadow-md"
   359→                  >
   360→                    <div className="relative h-24 w-full overflow-hidden">
   361→                      {r.image ? (
   362→                        <img
   363→                          src={r.image}
   364→                          alt={r.chineseName}
   365→                          className="h-full w-full object-cover transition-transform duration-500 group-hover:scale-110"
   366→                          loading="lazy"
   367→                        />
   368→                      ) : (
   369→                        <TaxaPlaceholder latinName={r.latinName} kingdom={taxon.kingdom} className="h-full w-full" />
   370→                      )}
   371→                      {r.conservation && (
   372→                        <span
   373→                          className={cn(
   374→                            "absolute right-1.5 top-1.5 rounded-sm px-1 py-0.5 text-[9px] font-bold text-white",
   375→                            IUCN_INFO[r.conservation]?.bg
   376→                          )}
   377→                        >
   378→                          {r.conservation}
   379→                        </span>
   380→                      )}
   381→                    </div>
   382→                    <div className="p-2">
   383→                      <p className="truncate text-[13px] font-semibold text-foreground group-hover:text-primary">
   384→                        {r.chineseName}
   385→                      </p>
   386→                      <p className="latin truncate text-[10px] text-muted-foreground">{r.latinName}</p>
   387→                    </div>
   388→                  </button>
   389→                ))}
   390→              </div>
   391→            </SectionCard>
   392→          )}
   393→
   394→          {/* 子单元 */}
   395→          {children.length > 0 && (
   396→            <SectionCard
   397→              title={
   398→                taxon.rank === "genus"
   399→                  ? `本属物种(${children.length})`
   400→                  : `下属阶元(${children.length})`
   401→              }
   402→              icon={Sparkles}
   403→            >
   404→              <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-3">
   405→                {children.map((c, i) => (
   406→                  <TaxonCard key={c.id} taxon={c} kingdom={taxon.kingdom} index={i} />
   407→                ))}
   408→              </div>
   409→            </SectionCard>
   410→          )}
   411→
   412→          {/* 上一/下一物种 */}
   413→          {(prev || next) && (
   414→            <div>
   415→              <p className="mb-2 flex items-center justify-between text-[11px] text-muted-foreground/70">
   416→                <span>同属相邻条目</span>
   417→                <span className="hidden items-center gap-1 sm:flex">
   418→                  <kbd className="rounded border border-foreground/15 bg-muted px-1.5 py-0.5 font-mono text-[10px]">←</kbd>
   419→                  <kbd className="rounded border border-foreground/15 bg-muted px-1.5 py-0.5 font-mono text-[10px]">→</kbd>
   420→                  键盘切换
   421→                </span>
   422→              </p>
   423→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   424→              {prev ? (
   425→                <button
   426→                  onClick={() => openTaxon(prev.id)}
   427→                  className="group flex items-center gap-3 rounded-xl border border-foreground/10 bg-card p-3.5 text-left shadow-sm transition-colors hover:border-primary/40"
   428→                >
   429→                  <ArrowLeft className="h-4 w-4 shrink-0 text-muted-foreground" />
   430→                  <span className="min-w-0">
   431→                    <span className="block text-[11px] text-muted-foreground">上一条</span>
   432→                    <span className="block truncate text-sm font-semibold text-foreground group-hover:text-primary">
   433→                      {prev.chineseName}
   434→                    </span>
   435→                    <span className="latin block truncate text-xs text-muted-foreground">{prev.latinName}</span>
   436→                  </span>
   437→                </button>
   438→              ) : (
   439→                <span />
   440→              )}
   441→              {next && (
   442→                <button
   443→                  onClick={() => openTaxon(next.id)}
   444→                  className="group flex items-center justify-end gap-3 rounded-xl border border-foreground/10 bg-card p-3.5 text-right shadow-sm transition-colors hover:border-primary/40"
   445→                >
   446→                  <span className="min-w-0">
   447→                    <span className="block text-[11px] text-muted-foreground">下一条</span>
   448→                    <span className="block truncate text-sm font-semibold text-foreground group-hover:text-primary">
   449→                      {next.chineseName}
   450→                    </span>
   451→                    <span className="latin block truncate text-xs text-muted-foreground">{next.latinName}</span>
   452→                  </span>
   453→                  <ArrowRight className="h-4 w-4 shrink-0 text-muted-foreground" />
   454→                </button>
   455→              )}
   456→              </div>
   457→            </div>

   459→        </div>
   460→
   461→        {/* 右列:侧栏 */}
