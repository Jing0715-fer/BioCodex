    16→function useCountUp(target: number, duration = 900) {
    17→  const [v, setV] = useState(0);
    18→  const ref = useRef<number>(0);
    19→  useEffect(() => {
    20→    const start = performance.now();
    21→    const from = ref.current;
    22→    let raf = 0;
    23→    const tick = (now: number) => {
    24→      const p = Math.min(1, (now - start) / duration);
    25→      const eased = 1 - Math.pow(1 - p, 3);
    26→      setV(Math.round(from + (target - from) * eased));
    27→      if (p < 1) raf = requestAnimationFrame(tick);
    28→      else ref.current = target;
    29→    };
    30→    raf = requestAnimationFrame(tick);
    31→    return () => cancelAnimationFrame(raf);
    32→  }, [target, duration]);
    33→  return v;
    34→}

    36→function StatCounter({ value, label, suffix }: { value: number; label: string; suffix?: string }) {
    37→  const v = useCountUp(value);
    38→  return (
    39→    <div className="flex flex-col items-center px-4 py-3 sm:items-start">
    40→      <span className="font-display text-3xl font-bold tabular-nums text-foreground sm:text-4xl">
    41→        {v.toLocaleString()}
    42→        {suffix}
    43→      </span>
    44→      <span className="mt-1 text-xs tracking-widest text-muted-foreground">{label}</span>
    45→    </div>
    46→  );
    47→}

    49→/** 环形进度指示(数据完备度徽章墙用) */
    50→function DataRing({
    51→  value, total, label, hint, color,
    52→}: { value: number; total: number; label: string; hint: string; color: string }) {
    53→  const pct = total > 0 ? value / total : 0;
    54→  const v = useCountUp(Math.round(pct * 100));
    55→  const R = 30;
    56→  const C = 2 * Math.PI * R;
    57→  return (
    58→    <div className="flex flex-col items-center gap-2.5 px-2 py-4 text-center">
    59→      <div className="relative h-[76px] w-[76px]">
    60→        <svg viewBox="0 0 76 76" className="h-full w-full -rotate-90">
    61→          <circle cx="38" cy="38" r={R} fill="none" strokeWidth="7" className="stroke-foreground/10" />
    62→          <circle
    63→            cx="38" cy="38" r={R} fill="none" strokeWidth="7" strokeLinecap="round"
    64→            stroke={color}
    65→            strokeDasharray={C}
    66→            strokeDashoffset={C * (1 - pct)}
    67→            style={{ transition: "stroke-dashoffset 1.1s cubic-bezier(.25,.1,.25,1)" }}
    68→          />
    69→        </svg>
    70→        <div className="absolute inset-0 flex flex-col items-center justify-center">
    71→          <span className="font-display text-lg font-bold tabular-nums leading-none text-foreground">{v}%</span>
    72→        </div>
    73→      </div>
    74→      <div>
    75→        <p className="text-xs font-bold tracking-wide text-foreground">{label}</p>
    76→        <p className="mt-0.5 text-[10px] tabular-nums text-muted-foreground">
    77→          {value} / {total} · {hint}
    78→        </p>
    79→      </div>
    80→    </div>
    81→  );
    82→}
