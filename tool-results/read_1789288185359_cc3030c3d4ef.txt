    53→
    54→export function TaxonDetail({ id }: { id: string }) {
    55→  const { openTaxon, explore, goBack, compareIds, toggleCompare, openCompare } = useBioStore();
    56→  const favorites = useFavorites();
    57→  const { data, isLoading } = useTaxon(id);
    58→  const { data: tree } = useTree();
    59→  const [lightbox, setLightbox] = useState(false);
    60→
    61→  // 面包屑物种计数:从树 DTO 递归建 id → 物种数映射(与探索视图一致)
    62→  const speciesCountById = useMemo(() => {
    63→    const m = new Map<string, number>();
    64→    const walk = (nodes: TreeNodeDTO[]) => {
    65→      for (const n of nodes) {
    66→        if (!m.has(n.id)) m.set(n.id, n.sc);
    67→        if (n.ch?.length) walk(n.ch);
    68→      }
    69→    };
    70→    if (tree) walk(tree);
    71→    return m;
    72→  }, [tree]);
    73→
    74→  const links = useMemo(
    75→    () =>
    76→      data
    77→        ? buildDbLinks({
    78→            latinName: data.taxon.latinName,
    79→            chineseName: data.taxon.chineseName,
    80→            ncbiTaxId: data.taxon.ncbiTaxId,
    81→            rank: data.taxon.rank,
    82→            kingdomPath: data.taxon.kingdomPath,
    83→          })
    84→        : [],
    85→    [data]
    86→  );
    87→
    88→  const grouped = useMemo(() => {
    89→    const g = new Map<string, typeof links>();
    90→    for (const l of links) {
    91→      if (!g.has(l.group)) g.set(l.group, []);
    92→      g.get(l.group)!.push(l);
    93→    }
    94→    return g;
    95→  }, [links]);
    96→
    97→  // 上一/下一物种(hooks 必须在 early return 之前计算)
    98→  const siblings = data?.siblings || [];
    99→  const idx = siblings.findIndex((s) => s.id === id);
   100→  const prev = idx > 0 ? siblings[idx - 1] : null;
   101→  const next = idx >= 0 && idx < siblings.length - 1 ? siblings[idx + 1] : null;
   102→
   103→  // 键盘导航:←/→ 切换同属上/下一物种(输入框聚焦时忽略);Esc 关灯箱
   104→  useEffect(() => {
   105→    const onKey = (e: KeyboardEvent) => {
   106→      const tag = (e.target as HTMLElement)?.tagName;
   107→      if (tag === "INPUT" || tag === "TEXTAREA" || (e.target as HTMLElement)?.isContentEditable) return;
   108→      if (e.key === "Escape") setLightbox(false);
   109→      if (e.key === "ArrowLeft" && prev) openTaxon(prev.id);
   110→      if (e.key === "ArrowRight" && next) openTaxon(next.id);
   111→      // F 键快捷收藏当前物种(仅物种阶元)
   112→      if (e.key.toLowerCase() === "f" && data?.success && data.taxon.rank === "species") {
   113→        const t = data.taxon;
   114→        const res = toggleFavorite({
   115→          id: t.id,
   116→          chineseName: t.chineseName,
   117→          latinName: t.latinName,
   118→          kingdom: t.kingdom,
   119→          image: t.image,
   120→          conservation: t.conservation,
   121→          ncbiTaxId: t.ncbiTaxId,
   122→          description: t.description,
   123→        });
   124→        if (res === "added") toast.success(`已收进标本夹:${t.chineseName}`, { description: "头栏书签图标可查看全部收藏" });
   125→        else if (res === "removed") toast.info(`已从标本夹取出:${t.chineseName}`);
   126→        else toast.warning(`标本夹已满(上限 ${FAVORITES_MAX} 件)`);
   127→      }
   128→    };
   129→    window.addEventListener("keydown", onKey);
   130→    return () => window.removeEventListener("keydown", onKey);
   131→  }, [prev, next, openTaxon, data]);
   132→
   133→  // 浏览足迹:查看任意条目时记录(localStorage 持久化,头栏「足迹」下拉可回溯)
   134→  useEffect(() => {
   135→    if (data?.success && data.taxon) {
   136→      pushHistory({
   137→        id: data.taxon.id,
   138→        chineseName: data.taxon.chineseName,
   139→        latinName: data.taxon.latinName,
   140→        kingdom: data.taxon.kingdom,
   141→        image: data.taxon.image,
   142→      });
   143→    }
   144→  }, [data]);
   145→
   146→  if (isLoading || !data?.success) {
   147→    return (
   148→      <div className="mx-auto max-w-[1400px] px-4 py-8 sm:px-6">
   149→        <div className="flex flex-col gap-4">
   150→          <Skeleton className="h-5 w-64" />
   151→          <Skeleton className="h-72 w-full rounded-xl" />
   152→          <div className="grid gap-4 lg:grid-cols-3">
   153→            <div className="space-y-4 lg:col-span-2">
   154→              <Skeleton className="h-40 w-full rounded-xl" />
   155→            </div>
   156→            <Skeleton className="h-40 rounded-xl" />
   157→          </div>
   158→        </div>
   159→      </div>
   160→    );
   161→  }
   162→
   163→  const { taxon, lineage, children, counts } = data;
   164→  const theme = KINGDOM_THEME[taxon.kingdom] || KINGDOM_THEME.Animalia;
   165→  const isSpecies = taxon.rank === "species";
   166→  const iucn = taxon.conservation ? IUCN_INFO[taxon.conservation] : null;
   167→  const tags = taxon.tags || [];
   168→  // 同属近亲(排除自身)
   169→  const relatedSpecies = siblings.filter(
   170→    (s) => s.rank === "species" && s.id !== taxon.id
   171→  );
   172→  const genusName = lineage[lineage.length - 1]?.chineseName;
   173→
   174→  return (
   175→    <div className="mx-auto w-full max-w-[1400px] px-4 pb-8 pt-4 sm:px-6">
   176→      {/* 面包屑 */}
   177→      <nav className="flex flex-wrap items-center gap-1 text-xs text-muted-foreground" aria-label="分类路径">
   178→        <button className="hover:text-primary" onClick={() => explore(null)}>
   179→          生命之树
   180→        </button>
   181→        {lineage.map((l) => {
   182→          const sc = speciesCountById.get(l.id);
   183→          return (
   184→            <span key={l.id} className="flex items-center gap-1">
   185→              <ChevronRight className="h-3 w-3 text-muted-foreground/50" />
   186→              <button
   187→                className="flex max-w-32 items-center gap-1 truncate hover:text-primary"
   188→                onClick={() => explore(l.id)}
   189→                title={`${l.chineseName}${sc ? ` · ${sc} 物种` : ""}`}
   190→              >
   191→                <span className="truncate">{l.chineseName}</span>
   192→                {sc != null && sc > 0 && (
   193→                  <span className="shrink-0 rounded-full bg-foreground/8 px-1.5 text-[9px] font-semibold tabular-nums text-muted-foreground/80 dark:bg-foreground/15">
   194→                    {sc}
   195→                  </span>
   196→                )}
   197→              </button>
   198→            </span>
   199→          );
   200→        })}
   201→        <ChevronRight className="h-3 w-3 text-muted-foreground/50" />
   202→        <span className="font-medium text-foreground">{taxon.chineseName}</span>
   203→      </nav>
   204→
   205→      {/* ====== 主图区 ====== */}
   206→      <div className="reveal-up relative mt-3 overflow-hidden rounded-2xl border border-foreground/10 shadow-md">
   207→        {taxon.image ? (
   208→          <button
   209→            onClick={() => setLightbox(true)}
   210→            aria-label="放大查看物种插图"
   211→            className="group/img absolute inset-0 h-full w-full cursor-zoom-in"
   212→          >
   213→            <img
   214→              src={taxon.image}
   215→              alt={`${taxon.chineseName}(${taxon.latinName})博物学插图`}
   216→              className="img-fade-in h-[320px] w-full object-cover transition-transform duration-500 group-hover/img:scale-[1.02] sm:h-[420px]"
   217→            />
   218→          </button>
   219→        ) : (
   220→          <TaxaPlaceholder latinName={taxon.latinName} kingdom={taxon.kingdom} big className="h-[320px] w-full sm:h-[420px]" />
   221→        )}
   222→        <div className="pointer-events-none absolute inset-0 bg-gradient-to-t from-black/70 via-black/15 to-transparent" />
   223→        {/* 界主题装饰纹样(博物馆钢印式,叠加于顶部左侧) */}
   224→        <KingdomOrnament
   225→          kingdom={taxon.kingdom}
   226→          className="absolute left-4 top-3 hidden h-8 w-36 text-white/40 drop-shadow-sm sm:block"
   227→        />
   228→        {/* 放大提示角标(有图时) */}
   229→        {taxon.image && (
   230→          <span className="pointer-events-none absolute bottom-3 right-3 flex items-center gap-1 rounded-full bg-black/50 px-2.5 py-1 text-[11px] font-medium text-white/90 opacity-0 backdrop-blur-sm transition-opacity duration-300 group-hover/img:opacity-100 sm:bottom-5 sm:right-5">
   231→            <ZoomIn className="h-3 w-3" />
   232→            点击放大
   233→          </span>
   234→        )}
   235→        {/* 右上角操作:加入对比 + 收藏(仅物种) */}
   236→        {isSpecies && (
   237→          <div className="absolute right-3 top-3 flex items-center gap-2">
   238→            {(() => {
   239→              const inCompare = compareIds.includes(taxon.id);
   240→              const full = !inCompare && compareIds.length >= MAX_COMPARE;
   241→              const isFav = favorites.some((x) => x.id === taxon.id);
   242→              return (
   243→                <>
   244→                  <button
   245→                    onClick={() => {
   246→                      const res = toggleFavorite({
   247→                        id: taxon.id,
   248→                        chineseName: taxon.chineseName,
   249→                        latinName: taxon.latinName,
   250→                        kingdom: taxon.kingdom,
   251→                        image: taxon.image,
   252→                        conservation: taxon.conservation,
   253→                        ncbiTaxId: taxon.ncbiTaxId,
   254→                        description: taxon.description,
   255→                      });
   256→                      if (res === "added")
   257→                        toast.success(`已收进标本夹:${taxon.chineseName}`, {
   258→                          description: `快捷键 F 也可随时收藏 · 头栏书签图标查看全部`,
   259→                        });
   260→                      else if (res === "removed") toast.info(`已从标本夹取出:${taxon.chineseName}`);
   261→                      else toast.warning(`标本夹已满(上限 ${FAVORITES_MAX} 件)`, { description: "可先清理一些不再需要的标本" });
   262→                    }}
   263→                    aria-label={isFav ? `从标本夹移除:${taxon.chineseName}` : `收藏到标本夹:${taxon.chineseName}`}
   264→                    className={cn(
   265→                      "flex items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-bold shadow transition-all",
   266→                      isFav
   267→                        ? "bg-amber-500 text-white"
   268→                        : "bg-black/50 text-white backdrop-blur-sm hover:bg-black/70"
   269→                    )}
   270→                  >
   271→                    <Bookmark className={cn("h-3.5 w-3.5", isFav && "fill-current")} />
   272→                    {isFav ? "已收藏" : "收藏"}
   273→                  </button>
   274→                  <button
   275→                    onClick={() => {
   276→                      const res = toggleCompare(taxon.id);
   277→                      if (res === "added")
   278→                        toast.success(`已加入对比:${taxon.chineseName}`, {
   279→                          description: `托盘 ${compareIds.length + 1}/${MAX_COMPARE},选满 2 个即可开始`,
   280→                        });
   281→                      else if (res === "removed") toast.info(`已移出对比:${taxon.chineseName}`);
   282→                      else toast.warning("对比托盘已满(最多 3 个)", { description: "可先开始对比或移除一个物种" });
   283→                    }}
   284→                    disabled={full}
   285→                    aria-label={inCompare ? `移出对比:${taxon.chineseName}` : `加入对比:${taxon.chineseName}`}
   286→                    className={cn(
   287→                      "flex items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-bold shadow transition-all",
   288→                      inCompare
   289→                        ? "bg-primary text-primary-foreground"
   290→                        : "bg-black/50 text-white backdrop-blur-sm hover:bg-black/70",
   291→                      full && "cursor-not-allowed opacity-50"
   292→                    )}
   293→                  >
   294→                    {inCompare ? <Check className="h-3.5 w-3.5" /> : <GitCompareArrows className="h-3.5 w-3.5" />}
   295→                    {inCompare ? "已加入对比" : "加入对比"}
   296→                  </button>
   297→                  {compareIds.length >= 2 && (
   298→                    <button
   299→                      onClick={openCompare}
   300→                      className="hidden items-center gap-1.5 rounded-full bg-white/20 px-3 py-1.5 text-xs font-bold text-white shadow backdrop-blur-sm transition-colors hover:bg-white/35 sm:flex"
   301→                    >
   302→                      对比 {compareIds.length} 个物种
   303→                    </button>
   304→                  )}
   305→                </>
   306→              );
   307→            })()}
   308→          </div>
   309→        )}
   310→        {/* 标题叠层 */}
   311→        <div className="absolute bottom-0 left-0 right-0 p-5 sm:p-8">
   312→          <div className="flex flex-wrap items-center gap-2">
   313→            <span
   314→              className="flex items-center gap-1.5 rounded-full px-3 py-1 text-xs font-bold text-white shadow"
   315→              style={{ background: theme.color }}
   316→            >
   317→              <KingdomIcon kingdom={taxon.kingdom} className="h-3.5 w-3.5" />
   318→              {theme.name}
   319→            </span>
   320→            <span className="rounded-sm bg-white/20 px-2 py-0.5 text-xs font-medium text-white backdrop-blur-sm">
   321→              {rankLabel(taxon.rank)}
   322→            </span>
   323→            {iucn && (
   324→              <span className={cn("rounded-sm px-2 py-0.5 text-xs font-bold text-white shadow", iucn.bg)}>
   325→                IUCN {iucn.label}
   326→              </span>
   327→            )}
   328→            {tags.includes("flagship") && (
   329→              <span className="flex items-center gap-1 rounded-sm bg-amber-500/85 px-2 py-0.5 text-xs font-bold text-white shadow">
   330→                <Star className="h-3 w-3" />
   331→                旗舰物种
   332→              </span>
   333→            )}
   334→            {tags.includes("模式生物") && (
   335→              <span className="flex items-center gap-1 rounded-sm bg-teal-600/85 px-2 py-0.5 text-xs font-bold text-white shadow">
   336→                <Microscope className="h-3 w-3" />
   337→                模式生物
   338→              </span>
   339→            )}
   340→            {tags.includes("入侵物种") && (
   341→              <span className="rounded-sm bg-zinc-700/85 px-2 py-0.5 text-xs font-bold text-white shadow">
   342→                入侵物种
   343→              </span>
   344→            )}
   345→          </div>
   346→          <h1 className="mt-3 font-display text-3xl font-bold text-white drop-shadow-md sm:text-5xl">
   347→            {taxon.chineseName}
   348→          </h1>
   349→          <div className="mt-1.5 flex flex-wrap items-baseline gap-3">
   350→            <span className="latin text-lg text-white/95 drop-shadow sm:text-2xl">{taxon.latinName}</span>
   351→            {taxon.authority && (
   352→              <span className="text-sm text-white/75 drop-shadow">{taxon.authority}</span>
   353→            )}
   354→            {taxon.ncbiTaxId && (
   355→              <span className="latin rounded-sm bg-white/15 px-2 py-0.5 text-xs text-white/90 backdrop-blur-sm">
   356→                NCBI txid{taxon.ncbiTaxId}
   357→              </span>
   358→            )}
   359→          </div>
   360→        </div>
   361→      </div>
   362→
   363→      {/* ====== 内容 + 侧栏 ====== */}
   364→      <div className="mt-5 grid gap-5 lg:grid-cols-[1fr_360px]">
   365→        {/* 左列:正文 */}
   366→        <div className="min-w-0 space-y-5">
   367→          {taxon.description && (
   368→            <SectionCard title={isSpecies ? "物种描述" : "类群概述"} icon={BookOpen}>
   369→              {taxon.description}
   370→            </SectionCard>
   371→          )}
   372→
   373→          {(taxon.morphology || taxon.habitat || taxon.distribution) && (
   374→            <SectionCard title="形态、生境与分布" icon={Leaf}>
   375→              <dl className="space-y-4">
   376→                {taxon.morphology && (
   377→                  <div>
   378→                    <dt className="mb-1 text-xs font-bold tracking-widest text-muted-foreground">形态识别</dt>
   379→                    <dd>{taxon.morphology}</dd>
   380→                  </div>
   381→                )}
   382→                {taxon.habitat && (
   383→                  <div>
   384→                    <dt className="mb-1 text-xs font-bold tracking-widest text-muted-foreground">生境</dt>
   385→                    <dd>{taxon.habitat}</dd>
   386→                  </div>
   387→                )}
   388→                {taxon.distribution && (
   389→                  <div>
   390→                    <dt className="mb-1 text-xs font-bold tracking-widest text-muted-foreground">分布</dt>
   391→                    <dd className="flex items-start gap-2">
   392→                      <MapPin className="mt-1.5 h-4 w-4 shrink-0 text-primary/70" />
   393→                      {taxon.distribution}
   394→                    </dd>
   395→                  </div>
   396→                )}
   397→              </dl>
   398→            </SectionCard>
   399→          )}
   400→
   401→          {iucn && taxon.conservation && (
   402→            <SectionCard title="保护状况" icon={Shield}>
   403→              <div className="flex flex-wrap items-center gap-3">
   404→                <span className={cn("rounded-md px-3 py-1.5 font-display text-lg font-bold text-white", iucn.bg)}>
   405→                  {taxon.conservation}
   406→                </span>
   407→                <div>
   408→                  <p className="font-semibold text-foreground">{iucn.full}</p>
   409→                  <p className="mt-0.5 text-xs leading-5 text-muted-foreground">
   410→                    依据 IUCN 红色名录等级:极危(CR)与濒危(EN)物种面临野外灭绝的高风险;
   411→                    图鉴中 {data.counts.speciesCount >= 0 && "收录的该等级物种均附红色名录直链,可查最新评估。"}
   412→                  </p>
   413→                </div>
   414→              </div>
   415→            </SectionCard>
   416→          )}
   417→
   418→          {/* 同属近亲 */}
   419→          {isSpecies && relatedSpecies.length > 0 && (
   420→            <SectionCard
   421→              title={`${genusName || "同属"}的其它成员(${relatedSpecies.length})`}
   422→              icon={Sparkles}
   423→            >
   424→              <div className="nh-scroll -mx-1 flex gap-3 overflow-x-auto px-1 pb-2">
   425→                {relatedSpecies.map((r) => (
   426→                  <button
   427→                    key={r.id}
   428→                    onClick={() => openTaxon(r.id)}
   429→                    className="group relative w-40 shrink-0 overflow-hidden rounded-lg border border-foreground/10 bg-muted/30 text-left transition-all hover:border-primary/40 hover:shadow-md"
   430→                  >
   431→                    <div className="relative h-24 w-full overflow-hidden">
   432→                      {r.image ? (
   433→                        <img
   434→                          src={r.image}
   435→                          alt={r.chineseName}
   436→                          className="h-full w-full object-cover transition-transform duration-500 group-hover:scale-110"
   437→                          loading="lazy"
   438→                        />
   439→                      ) : (
   440→                        <TaxaPlaceholder latinName={r.latinName} kingdom={taxon.kingdom} className="h-full w-full" />
   441→                      )}
   442→                      {r.conservation && (
   443→                        <span
   444→                          className={cn(
   445→                            "absolute right-1.5 top-1.5 rounded-sm px-1 py-0.5 text-[9px] font-bold text-white",
